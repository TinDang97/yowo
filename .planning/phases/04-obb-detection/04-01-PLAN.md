    def dist2rbox(
        pred_dist: Tensor,    # (B, 4, A)
        pred_angle: Tensor,   # (B, 1, A)
        anchor_points: Tensor,  # (A, 2)
        *,
        anchors_t: Tensor | None = None,
    ) -> Tensor:
        """Decode DFL distances + rotation angle to (cx, cy, w, h) rotated box."""
        import math as _math  # math already imported at module level via 'import math'
        lt, rb = pred_dist.chunk(2, dim=1)   # each (B, 2, A)
        cos = torch.cos(pred_angle)           # (B, 1, A)
        sin = torch.sin(pred_angle)
        xf = (rb[:, 0:1] - lt[:, 0:1]) * 0.5
        yf = (rb[:, 1:2] - lt[:, 1:2]) * 0.5
        if anchors_t is None:
            anchors_t = anchor_points.T.unsqueeze(0)  # (1, 2, A)
        cx = xf * cos - yf * sin + anchors_t[:, 0:1]
        cy = xf * sin + yf * cos + anchors_t[:, 1:2]
        w = lt[:, 0:1] + rb[:, 0:1]
        h = lt[:, 1:2] + rb[:, 1:2]
        return torch.cat([cx, cy, w, h], dim=1)  # (B, 4, A)
    ```

    class OBBHead(Detect):
        """OBB detection head: adds angle branch cv4 alongside cv2 (box) and cv3 (cls).
        Output: (B, 4+nc+ne, total_anchors) at inference. Default nc=15 for DOTA v1.
        """
        def __init__(self, nc: int = 15, ne: int = 1, reg_max: int = 16, ch: tuple[int,...] = ()) -> None:
            super().__init__(nc=nc, reg_max=reg_max, end2end=False, ch=ch)
            self.ne = ne
            c4 = max(ch[0] // 4, self.ne) if ch else self.ne
            self.cv4 = nn.ModuleList(
                nn.Sequential(Conv(c, c4, 3), Conv(c4, c4, 3), nn.Conv2d(c4, self.ne, 1))
                for c in ch
            )

        def forward(self, x: list[Tensor]) -> Tensor:
            import math
            box_feats = [self.cv2[i](x[i]) for i in range(self.nl)]
            cls_feats = [self.cv3[i](x[i]) for i in range(self.nl)]
            angle_feats = [self.cv4[i](x[i]) for i in range(self.nl)]
            box_cat = torch.cat([b.flatten(2) for b in box_feats], dim=2)   # (B, 4*reg_max, A)
            cls_cat = torch.cat([c.flatten(2) for c in cls_feats], dim=2)   # (B, nc, A)
            ang_cat = torch.cat([a.flatten(2) for a in angle_feats], dim=2)  # (B, ne, A)
            dfl_out = self.dfl(box_cat)
            ang_cat = (ang_cat.sigmoid() - 0.25) * math.pi  # [-pi/4, 3pi/4] — applied ONCE
            # Initialize strides+anchor cache on first forward (same mechanism as Detect)
            if not hasattr(self, '_cached_anchors') or self._cached_anchors is None:
                self._init_strides([x[i] for i in range(self.nl)])
            dbox = dist2rbox(dfl_out, ang_cat, self._cached_anchors) * self._cached_strides_t
            cls_cat = cls_cat.sigmoid()
            return torch.cat([dbox, cls_cat, ang_cat], dim=1)  # (B, 4+nc+ne, A)
    ```

    ```python
    """OBB postprocessing: probiou NMS and OBBDetection assembly."""
    from __future__ import annotations
    import numpy as np
    import torch
    from torch import Tensor
    from numpy.typing import NDArray
    from yowo.types import Frame, ModelSpec, OBBBox, OBBDetection, PreprocessedTensor

    DOTA_CLASSES: list[str] = [
        "plane", "ship", "storage tank", "baseball diamond", "tennis court",
        "basketball court", "ground track field", "harbor", "bridge",
        "large vehicle", "small vehicle", "helicopter", "roundabout",
        "soccer ball field", "swimming pool",
    ]

    def _get_covariance_matrix(boxes: Tensor) -> tuple[Tensor, Tensor, Tensor]:
        """Covariance components from (N,5) xywhr boxes."""
        gbbs = torch.cat((boxes[:, 2:4].pow(2) / 12, boxes[:, 4:5]), dim=-1)
        a, b, c_angle = gbbs.unbind(-1)
        cos, sin = torch.cos(c_angle), torch.sin(c_angle)
        cos2, sin2, sincos = cos.pow(2), sin.pow(2), cos * sin
        return a * cos2 + b * sin2, a * sin2 + b * cos2, (a - b) * sincos

    def probiou_matrix(obb1: Tensor, obb2: Tensor, eps: float = 1e-7) -> Tensor:
        """(N,5) x (M,5) xywhr tensors -> (N,M) IoU matrix via Gaussian probiou."""
        x1, y1 = obb1[:, 0:1], obb1[:, 1:2]    # (N,1)
        x2, y2 = obb2[:, 0:1].T, obb2[:, 1:2].T  # (1,M)
        a1, b1, c1 = _get_covariance_matrix(obb1)  # (N,)
        a2, b2, c2 = _get_covariance_matrix(obb2)  # (M,)
        a1, b1, c1 = a1[:, None], b1[:, None], c1[:, None]  # (N,1)
        a2, b2, c2 = a2[None, :], b2[None, :], c2[None, :]  # (1,M)
        t1 = ((a1+a2)*(y1-y2)**2 + (b1+b2)*(x1-x2)**2) / (
            (a1+a2)*(b1+b2) - (c1+c2)**2 + eps) * 0.25
        t2 = ((c1+c2)*(x2-x1)*(y1-y2)) / (
            (a1+a2)*(b1+b2) - (c1+c2)**2 + eps) * 0.5
        t3 = (((a1+a2)*(b1+b2) - (c1+c2)**2) / (
            4 * ((a1*b1 - c1**2).clamp_(0) * (a2*b2 - c2**2).clamp_(0)).sqrt() + eps
        ) + eps).log() * 0.5
        bd = (t1 + t2 + t3).clamp(eps, 100.0)
        hd = (1.0 - (-bd).exp() + eps).sqrt()
        return 1.0 - hd

    def _nms_rotated(
        boxes_xywhr: Tensor,  # (N, 5)
        scores: Tensor,       # (N,)
        iou_threshold: float,
    ) -> Tensor:
        """Greedy NMS using probiou. Returns kept indices."""
        order = scores.argsort(descending=True)
        kept: list[int] = []
        suppressed = torch.zeros(len(order), dtype=torch.bool, device=boxes_xywhr.device)
        for i in range(len(order)):
            idx = int(order[i].item())
            if suppressed[idx]:
                continue
            kept.append(idx)
            if i + 1 >= len(order):
                break
            rest_idx = order[i+1:]
            rest_idx = rest_idx[~suppressed[rest_idx]]
            if len(rest_idx) == 0:
                break
            iou = probiou_matrix(boxes_xywhr[idx:idx+1], boxes_xywhr[rest_idx]).squeeze(0)
            suppress_mask = iou > iou_threshold
            suppressed[rest_idx[suppress_mask]] = True
        return torch.tensor(kept, dtype=torch.long, device=boxes_xywhr.device)

    def postprocess_obb(
        raw: Tensor,            # (B, 4+nc+1, A) from OBBHead.forward()
        frames: list[Frame],
        spec: ModelSpec,
        conf_threshold: float = 0.25,
        iou_threshold: float = 0.45,
        class_names: list[str] | None = None,
    ) -> list[OBBDetection]:
        """Convert raw OBBHead output to list of OBBDetection (one per frame)."""
        names = class_names or DOTA_CLASSES
        nc = spec.num_classes or 15
        results: list[OBBDetection] = []
        for b in range(raw.shape[0]):
            pred = raw[b]  # (4+nc+1, A)
            boxes_xywh = pred[:4].T          # (A, 4)
            cls_scores  = pred[4:4+nc].T     # (A, nc)
            angles      = pred[4+nc:].T      # (A, 1)
            # Per-anchor max class confidence
            conf, cls_ids = cls_scores.max(dim=1)  # (A,), (A,)
            mask = conf > conf_threshold
            if not mask.any():
                results.append(OBBDetection(
                    frame_index=frames[b].frame_index,
                    source_id=frames[b].source_id,
                    boxes=(),
                ))
                continue
            boxes_f = boxes_xywh[mask]     # (K, 4)
            angles_f = angles[mask, 0]     # (K,)
            conf_f   = conf[mask]          # (K,)
            cls_ids_f = cls_ids[mask]      # (K,)
            # Class-aware offset trick for NMS separation
            class_offsets = cls_ids_f.float().unsqueeze(1) * 10000.0
            boxes_offset = boxes_f.clone()
            boxes_offset[:, :2] += class_offsets
            xywhr = torch.cat([boxes_offset, angles_f.unsqueeze(1)], dim=1)  # (K, 5)
            kept = _nms_rotated(xywhr, conf_f, iou_threshold)
            obb_boxes: list[OBBBox] = []
            for k in kept.tolist():
                cid = int(cls_ids_f[k].item())
                obb_boxes.append(OBBBox(
                    cx=float(boxes_f[k, 0].item()),
                    cy=float(boxes_f[k, 1].item()),
                    w=float(boxes_f[k, 2].item()),
                    h=float(boxes_f[k, 3].item()),
                    angle=float(angles_f[k].item()),
                    confidence=float(conf_f[k].item()),
                    class_id=cid,
                    class_name=names[cid] if cid < len(names) else "",
                ))
            results.append(OBBDetection(
                frame_index=frames[b].frame_index,
                source_id=frames[b].source_id,
                boxes=tuple(obb_boxes),
            ))
        return results

    __all__ = ["DOTA_CLASSES", "postprocess_obb", "probiou_matrix"]
    ```
